cmake_minimum_required(VERSION 3.20)
project(LikeEdit VERSION 1.0.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
include(CTest)

add_library(likeedit_core STATIC src/editor_core.c src/syntax.c)
target_include_directories(likeedit_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
if(MSVC)
    target_compile_options(likeedit_core PRIVATE /W4)
else()
    target_compile_options(likeedit_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(WIN32)
    add_executable(LikeEdit src/win_editor.c)
    target_compile_definitions(LikeEdit PRIVATE UNICODE _UNICODE)
    target_link_libraries(LikeEdit PRIVATE likeedit_core)
    if(MSVC)
        target_compile_options(LikeEdit PRIVATE /W4)
    elseif(MINGW)
        target_link_options(LikeEdit PRIVATE -municode)
    endif()
endif()

if(BUILD_TESTING)
    add_executable(likeedit_tests tests/test_core.c)
    target_link_libraries(likeedit_tests PRIVATE likeedit_core)
    add_test(NAME EditorCore COMMAND likeedit_tests)
endif()
